.version 8.0
.target sm_75
.address_size 64

.visible .entry cuda_js_vector_add_u32(
    .param .u64 output_pointer,
    .param .u64 left_pointer,
    .param .u64 right_pointer,
    .param .u32 element_count
)
{
    .reg .pred %p<2>;
    .reg .b32 %r<8>;
    .reg .b64 %rd<10>;

    mov.u32 %r1, %ctaid.x;
    mov.u32 %r2, %ntid.x;
    mov.u32 %r3, %tid.x;
    mad.lo.s32 %r4, %r1, %r2, %r3;
    ld.param.u32 %r5, [element_count];
    setp.ge.u32 %p1, %r4, %r5;
    @%p1 bra complete;

    ld.param.u64 %rd1, [output_pointer];
    ld.param.u64 %rd2, [left_pointer];
    ld.param.u64 %rd3, [right_pointer];
    mul.wide.u32 %rd4, %r4, 4;
    add.s64 %rd5, %rd2, %rd4;
    add.s64 %rd6, %rd3, %rd4;
    add.s64 %rd7, %rd1, %rd4;
    ld.global.u32 %r6, [%rd5];
    ld.global.u32 %r7, [%rd6];
    add.u32 %r6, %r6, %r7;
    st.global.u32 [%rd7], %r6;

complete:
    ret;
}
