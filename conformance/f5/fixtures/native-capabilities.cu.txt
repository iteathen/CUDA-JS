#include <stdint.h>
#include <cuda/atomic>

extern "C" __global__ void cuda_js_native_scalar(
    uint32_t legacy,
    unsigned long long wide,
    int32_t signed_value,
    float fractional,
    uint32_t *output) {
  if (blockIdx.x != 0 || threadIdx.x != 0) return;
  output[0] = legacy;
  output[1] = (uint32_t)wide;
  output[2] = (uint32_t)(wide >> 32);
  output[3] = (uint32_t)signed_value;
  output[4] = __float_as_uint(fractional);
}

extern "C" __global__ void cuda_js_native_delayed(
    uint32_t *output,
    unsigned long long cycles) {
  if (blockIdx.x != 0 || threadIdx.x != 0) return;
  const unsigned long long started = clock64();
  while (clock64() - started < cycles) {
  }
  output[0] = UINT32_C(0xc001d00d);
}

extern "C" __global__ void cuda_js_native_deferred_fault(uint32_t *output) {
  if (blockIdx.x != 0 || threadIdx.x != 0) return;
  output[0] = UINT32_C(0x12345678);
  asm volatile("trap;");
}

extern "C" __global__ void cuda_js_native_atomic_producer(
    uint32_t *shared_value,
    unsigned long long cycles) {
  if (blockIdx.x != 0 || threadIdx.x != 0) return;
  atomicExch(shared_value, UINT32_C(1));
  const unsigned long long started = clock64();
  while (clock64() - started < cycles) {
  }
  atomicExch(shared_value, UINT32_C(2));
}

extern "C" __global__ void cuda_js_native_atomic_observer(
    uint32_t *shared_value,
    uint32_t *observation) {
  if (blockIdx.x != 0 || threadIdx.x != 0) return;
  observation[0] = atomicAdd(shared_value, UINT32_C(0));
}

extern "C" __global__ void cuda_js_native_transfer_increment(
    uint32_t *values,
    uint32_t word_count) {
  const uint32_t index = blockIdx.x * blockDim.x + threadIdx.x;
  if (index < word_count) values[index] += UINT32_C(1);
}

extern "C" __global__ void cuda_js_native_mailbox(
    uint32_t *host_to_device,
    uint32_t *device_to_host) {
  if (blockIdx.x != 0 || threadIdx.x != 0) return;
  cuda::atomic_ref<uint32_t, cuda::thread_scope_system> input(*host_to_device);
  cuda::atomic_ref<uint32_t, cuda::thread_scope_system> output(*device_to_host);
  uint32_t value = 0;
  while (value == 0) value = input.load(cuda::memory_order_acquire);
  output.store(value + UINT32_C(1), cuda::memory_order_release);
}
